// Calcola area e perimetro
public class Cerchio {
  public static void main(
      String[] args) {
    double r = 5.0;
    double area =
      Math.PI * r * r;
    double perim =
      2 * Math.PI * r;
    System.out.printf(
      "Area = %.2f%n", area);
    System.out.printf(
      "Perimetro = %.2f%n", perim);
  }
}
